int led1 = 10;
int led2 = 11;
int potn = A1;
void setup()
{
  pinMode(led1, OUTPUT);
  pinMode(led2, OUTPUT);
  Serial.begin(9600);
  
}

void loop()
{
  int potnValue = analogRead(potn);
  //Serial.println(potnValue);
  
  
  int brightness = map(potnValue, 0, 1023, 0, 255);
  
  Serial.println(brightness);
  analogWrite(led1, brightness);
  
  if(brightness > 180)
  { 
    digitalWrite(led2, HIGH);
  }
  else
  {
    digitalWrite(led2, LOW);
  }
}
