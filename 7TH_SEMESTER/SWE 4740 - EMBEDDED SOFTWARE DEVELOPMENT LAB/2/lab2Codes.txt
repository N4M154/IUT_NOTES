int LED = 8;
int BUTTON = 4;
void setup()
{
  pinMode(LED, OUTPUT);
  pinMode(BUTTON, INPUT);
  //in case of pullup method you write INPUT_PULLUP
}

void loop()
{
   int buttonState = digitalRead(BUTTON);
  
   if(buttonState == HIGH)
   {
     digitalWrite(LED,HIGH);
   }
  else
  {
    digitalWrite(LED,LOW);
  }
                            
}

int LED = 8;
int BUTTON = 4;

int ledState = LOW;          
int lastButtonPressState = LOW;   

void setup()
{
  pinMode(LED, OUTPUT);
  pinMode(BUTTON, INPUT); 
  // when using pull up method it's INPUT_PULLUP);
}

void loop()
{
  int buttonState = digitalRead(BUTTON);


  if (buttonState == HIGH && lastButtonPressState == LOW)
  {
    ledState = !ledState;         
    digitalWrite(LED, ledState);
                       
  }

  lastButtonPressState = buttonState; 
}
